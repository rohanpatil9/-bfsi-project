def norm(value: Any) -> str:
    return re.sub(r"\s+", " ", str(value or "").replace("\u00a0", " ")).strip()


def sid(*values: Any) -> str:
    return hashlib.sha1("|".join(map(str, values)).encode("utf-8")).hexdigest()


def parse_number(value: Any) -> float | None:
    if value is None:
        return None
    s = norm(value)
    if not s or s.upper() in {"-", "—", "–", "N/A", "NM"}:
        return None

    negative = s.startswith("(") and s.endswith(")")
    s = (
        s.replace(",", "")
        .replace("$", "")
        .replace("€", "")
        .replace("£", "")
        .replace("%", "")
        .strip()
        .strip("()")
    )

    try:
        number = float(s)
    except ValueError:
        return None

    return -number if negative else number


def detect_unit(rows: list[list[str]]) -> str:
    table_text = " ".join(
        norm(cell) for row in rows for cell in row
    ).lower()

    if "in millions" in table_text:
        return "USD millions"
    if "in billions" in table_text:
        return "USD billions"
    if "%" in table_text:
        return "percent"
    return "unknown"


def duration_from_text(text: str) -> int | None:
    text = norm(text).lower()

    if "three months ended" in text:
        return 3
    if "six months ended" in text:
        return 6
    if "nine months ended" in text:
        return 9
    if "twelve months ended" in text:
        return 12

    return None


def period_label(year: int, fiscal_quarter: str, duration_months: int) -> str:
    if duration_months == 3:
        return f"{year}-{fiscal_quarter}"
    if duration_months == 6:
        return f"{year}-H1"
    if duration_months == 9:
        return f"{year}-9M"
    if duration_months == 12:
        return f"{year}-FY"
    return f"{year}-{fiscal_quarter}-{duration_months}M"


def detect_period_columns(
    rows: list[list[str]],
    fiscal_quarter: str,
) -> dict[int, dict[str, Any]]:
    """
    Detect year/duration for each physical table column.

    This intentionally uses the extracted table structure instead of
    assuming that the final two numeric cells are the desired values.
    """

    result: dict[int, dict[str, Any]] = {}
    current_duration: int | None = None

    for row in rows:
        row_text = " ".join(norm(x) for x in row if norm(x))
        row_duration = duration_from_text(row_text)

        if row_duration is not None:
            current_duration = row_duration

        for column_index, cell in enumerate(row):
            cell_text = norm(cell)

            if not re.fullmatch(r"20\d{2}", cell_text):
                continue

            # Conservative fallback for ordinary quarterly tables.
            duration = current_duration or 3

            result[column_index] = {
                "year": int(cell_text),
                "duration_months": duration,
            }

    return result


def extract_period_values(
    rows: list[list[str]],
    row: list[str],
    fiscal_quarter: str,
) -> list[dict[str, Any]]:

    period_columns = detect_period_columns(rows, fiscal_quarter)
    output = []

    for column_index, cell in enumerate(row):
        if column_index == 0:
            continue

        value = parse_number(cell)
        if value is None:
            continue

        header = period_columns.get(column_index)
        if not header:
            continue

        output.append({
            "period": period_label(
                header["year"],
                fiscal_quarter,
                header["duration_months"],
            ),
            "value": value,
            "duration_months": header["duration_months"],
        })

    return output


def parse_file(path: Path) -> tuple[str, int, str]:
    match = re.search(r"FY(\d+)_Q([123])_10Q", path.stem, re.IGNORECASE)
    if not match:
        raise ValueError(f"Bad filename: {path}")

    company = path.parent.name.lower()
    if company not in COMPANIES:
        raise ValueError(f"Unknown company directory: {company}")

    return company, int(match.group(1)), f"Q{match.group(2)}"


def detect_section(text: str) -> str | None:
    patterns = [
        r"ITEM\s+\d+[A-Z]?\s*\.?.{0,120}",
        r"CONSOLIDATED STATEMENTS? OF [A-Z ]+",
        r"NOTES TO CONSOLIDATED FINANCIAL STATEMENTS",
        r"MANAGEMENT['’]S DISCUSSION AND ANALYSIS.{0,120}",
    ]

    for pattern in patterns:
        match = re.search(pattern, text, re.IGNORECASE)
        if match:
            return norm(match.group(0))[:180]

    return None


def extract_tables(page: fitz.Page) -> list[tuple[int, list[list[str]]]]:
    output = []

    try:
        for index, table in enumerate(page.find_tables().tables):
            rows = table.extract()
            clean_rows = [
                [norm(cell) for cell in row]
                for row in rows
                if any(norm(cell) for cell in row)
            ]
            if clean_rows:
                output.append((index, clean_rows))
    except Exception:
        pass

    return output


def extract_pdf(
    pdf: Path,
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:

    company, fiscal_year, fiscal_quarter = parse_file(pdf)
    period_end = COMPANIES[company][1][fiscal_quarter]

    document = fitz.open(pdf)
    chunks = []
    facts = []

    for page_number, page in enumerate(document, start=1):
        text = norm(page.get_text("text"))
        section = detect_section(text)
        parts = []

        if text:
            parts.append(("text", None, text))

        for table_index, rows in extract_tables(page):
            table_text = "\n".join(" | ".join(row) for row in rows)
            table_name = f"table_{table_index}"
            parts.append(("table", table_name, table_text))

            unit = detect_unit(rows)

            for row in rows:
                if not row:
                    continue

                source_label = norm(row[0])
                label_lower = source_label.lower()

                metric = next(
                    (
                        name
                        for name, aliases in ALIASES.items()
                        if any(alias in label_lower for alias in aliases)
                    ),
                    None,
                )

                if not metric:
                    continue

                period_values = extract_period_values(
                    rows,
                    row,
                    fiscal_quarter,
                )

                if not period_values:
                    continue

                facts.append({
                    "company": company,
                    "fiscal_year": fiscal_year,
                    "fiscal_quarter": fiscal_quarter,
                    "period_end": period_end,
                    "metric": metric,
                    "source_label": source_label,
                    "unit": unit,
                    "period_values": period_values,
                    "pdf_page": page_number,
                    "section": section,
                    "table_index": table_index,
                    "source_file": pdf.name,
                    "created_at": datetime.now(timezone.utc).isoformat(),
                })

        for _, (chunk_type, table_name, content) in enumerate(parts):
            if not content:
                continue

            if len(content) <= 6000 or chunk_type == "table":
                pieces = [content]
            else:
                pieces = [
                    content[i:i + 5500]
                    for i in range(0, len(content), 5000)
                ]

            for piece_index, piece in enumerate(pieces):
                chunks.append({
                    "chunk_id": sid(
                        company,
                        fiscal_year,
                        fiscal_quarter,
                        page_number,
                        chunk_type,
                        table_name,
                        piece_index,
                    ),
                    "company": company,
                    "fiscal_year": fiscal_year,
                    "fiscal_quarter": fiscal_quarter,
                    "period_end": period_end,
                    "pdf_page": page_number,
                    "section": section,
                    "table_name": table_name,
                    "chunk_type": chunk_type,
                    "source_file": pdf.name,
                    "content": piece,
                })

    document.close()
    return chunks, facts

def insert_rows_batched(
    table_id: str,
    rows: list[dict[str, Any]],
    batch_size: int,
) -> None:

    client = bq()

    for start in range(0, len(rows), batch_size):
        batch = rows[start:start + batch_size]
        errors = client.insert_rows_json(table_id, batch)

        if errors:
            raise RuntimeError(
                f"BigQuery insert failed for {table_id}: {errors}"
            )

def facts(
    company: str,
    fiscal_year: int,
    fiscal_quarter: str,
    metric: str,
) -> list[dict[str, Any]]:

    client = bq()

    sql = f"""
    SELECT *
    FROM `{PROJECT}.{DATASET}.{FACTS_V2}`
    WHERE LOWER(company) = @company
      AND fiscal_year = @fiscal_year
      AND UPPER(fiscal_quarter) = @fiscal_quarter
      AND metric = @metric
    ORDER BY pdf_page
    """

    params = [
        bigquery.ScalarQueryParameter("company", "STRING", company.lower()),
        bigquery.ScalarQueryParameter("fiscal_year", "INT64", fiscal_year),
        bigquery.ScalarQueryParameter(
            "fiscal_quarter", "STRING", fiscal_quarter.upper()
        ),
        bigquery.ScalarQueryParameter("metric", "STRING", metric),
    ]

    return [
        dict(row)
        for row in client.query(
            sql,
            job_config=bigquery.QueryJobConfig(query_parameters=params),
        ).result()
    ]


def get_period_value(
    fact: dict[str, Any],
    period: str,
    duration_months: int | None = None,
) -> float | None:

    for item in fact.get("period_values", []):
        if item.get("period") != period:
            continue

        if (
            duration_months is not None
            and item.get("duration_months") != duration_months
        ):
            continue

        if item.get("value") is not None:
            return float(item["value"])

    return None


def find_fact_for_period(
    fact_rows: list[dict[str, Any]],
    period: str,
    duration_months: int | None = None,
) -> tuple[float | None, dict[str, Any] | None]:

    for fact in fact_rows:
        value = get_period_value(
            fact,
            period,
            duration_months,
        )

        if value is not None:
            return value, fact

    return None, None


def calculate_metric_yoy(
    company: str,
    fiscal_year: int,
    fiscal_quarter: str,
    metric: str,
) -> dict[str, Any] | None:

    current_period = f"{fiscal_year}-{fiscal_quarter}"
    previous_period = f"{fiscal_year - 1}-{fiscal_quarter}"

    rows = facts(
        company,
        fiscal_year,
        fiscal_quarter,
        metric,
    )

    current, current_fact = find_fact_for_period(
        rows,
        current_period,
        3,
    )

    previous, previous_fact = find_fact_for_period(
        rows,
        previous_period,
        3,
    )

    if current is None or previous is None:
        return None

    if previous == 0:
        return None

    growth = round(
        ((current - previous) / previous) * 100,
        2,
    )

    return {
        "metric": metric,
        "current_period": current_period,
        "previous_period": previous_period,
        "current_value": current,
        "previous_value": previous,
        "growth_percent": growth,
        "unit": current_fact.get("unit") if current_fact else "unknown",
        "current_source": current_fact,
        "previous_source": previous_fact,
        "formula": "(current - previous) / previous * 100",
    }

def embed(
    texts: list[str],
    task: str = "RETRIEVAL_DOCUMENT",
) -> list[list[float]]:

    client = genai_client()
    output = []

    for start in range(0, len(texts), EMBED_BATCH):
        batch = texts[start:start + EMBED_BATCH]

        response = client.models.embed_content(
            model=EMBED_MODEL,
            contents=batch,
            config=types.EmbedContentConfig(
                task_type=task,
                output_dimensionality=EMBED_DIM,
                auto_truncate=True,
            ),
        )

        output.extend(
            [list(embedding.values) for embedding in response.embeddings]
        )

    return output


def search(
    question: str,
    company: str | None = None,
    fiscal_year: int | None = None,
    fiscal_quarter: str | None = None,
) -> list[dict[str, Any]]:

    query_embedding = embed(
        [question],
        "RETRIEVAL_QUERY",
    )[0]

    client = bq()
    where = []

    params = [
        bigquery.ArrayQueryParameter(
            "embedding", "FLOAT64", query_embedding
        ),
        bigquery.ScalarQueryParameter(
            "question", "STRING", question
        ),
        bigquery.ScalarQueryParameter(
            "topk", "INT64", TOP_K
        ),
    ]

    if company:
        where.append("company = @company")
        params.append(
            bigquery.ScalarQueryParameter(
                "company", "STRING", company.lower()
            )
        )

    if fiscal_year:
        where.append("fiscal_year = @fy")
        params.append(
            bigquery.ScalarQueryParameter(
                "fy", "INT64", fiscal_year
            )
        )

    if fiscal_quarter:
        where.append("fiscal_quarter = @q")
        params.append(
            bigquery.ScalarQueryParameter(
                "q", "STRING", fiscal_quarter.upper()
            )
        )

    base = f"""
        SELECT *
        FROM `{PROJECT}.{DATASET}.{CHUNKS}`
    """

    if where:
        base += " WHERE " + " AND ".join(where)

    sql = f"""
    SELECT
        base.chunk_id,
        base.company,
        base.fiscal_year,
        base.fiscal_quarter,
        base.period_end,
        base.pdf_page,
        base.section,
        base.table_name,
        base.chunk_type,
        base.source_file,
        base.content,
        distance
    FROM VECTOR_SEARCH(
        ({base}),
        'embedding',
        query_value => @embedding,
        lexical_search_columns => ['content'],
        lexical_search_query_value => @question,
        top_k => @topk,
        distance_type => 'COSINE'
    )
    ORDER BY distance
    """

    return [
        dict(row)
        for row in client.query(
            sql,
            job_config=bigquery.QueryJobConfig(
                query_parameters=params
            ),
        ).result()
    ]

def intent(question: str) -> dict[str, Any]:

    client = genai_client()

    prompt = f"""
Classify this question about SEC 10-Q financial filings.

Return only JSON matching the supplied schema.

Rules:
- Never invent a company, year or quarter.
- metrics may contain more than one metric.
- Use "comparison" when the user compares companies or metrics.
- Use "calculation" when arithmetic is required.
- Use "compare_growth" when comparing growth rates of two metrics.
- Supported metrics:
  revenue
  research_and_development
  operating_income
  net_income

Question:
{question}
"""

    response = client.models.generate_content(
        model=MODEL,
        contents=prompt,
        config=types.GenerateContentConfig(
            temperature=0,
            response_mime_type="application/json",
            response_schema=INTENT_SCHEMA,
        ),
    )

    return json.loads(response.text)


def calculate(
    intent_result: dict[str, Any],
) -> list[dict[str, Any]]:

    company = intent_result.get("company")
    fiscal_year = intent_result.get("fiscal_year")
    fiscal_quarter = intent_result.get("fiscal_quarter")
    metrics = intent_result.get("metrics") or []

    if not all([company, fiscal_year, fiscal_quarter, metrics]):
        return []

    calculation_type = intent_result.get(
        "calculation",
        "none",
    )

    results = []

    if calculation_type in {
        "yoy_growth",
        "compare_growth",
    }:
        for metric in metrics:
            result = calculate_metric_yoy(
                company,
                int(fiscal_year),
                fiscal_quarter,
                metric,
            )

            if result:
                results.append(result)

    if (
        calculation_type == "compare_growth"
        and len(results) >= 2
    ):
        first = results[0]
        second = results[1]

        difference = round(
            first["growth_percent"]
            - second["growth_percent"],
            2,
        )

        results.append({
            "type": "growth_rate_comparison",
            "metric_a": first["metric"],
            "growth_a": first["growth_percent"],
            "metric_b": second["metric"],
            "growth_b": second["growth_percent"],
            "difference_percentage_points": difference,
            "formula": "growth_a - growth_b",
        })

    return results


def answer(
    question: str,
    evidence: list[dict[str, Any]],
    calculations: list[dict[str, Any]],
) -> str:

    client = genai_client()

    evidence_blocks = []

    for number, item in enumerate(evidence, start=1):
        evidence_blocks.append(
            f"""
[S{number}]
Company: {item["company"]}
Fiscal year: {item["fiscal_year"]}
Fiscal quarter: {item["fiscal_quarter"]}
PDF page: {item["pdf_page"]}
Section: {item.get("section")}
Table: {item.get("table_name")}
Source file: {item["source_file"]}

{item["content"]}
"""
        )

    prompt = f"""
You are a financial analyst assistant.

Answer the user's question using ONLY the supplied evidence.

Rules:
1. Never invent financial numbers.
2. Deterministic calculations are authoritative.
3. Do not perform arithmetic yourself when a calculation result is supplied.
4. Every factual financial claim must be supported by evidence.
5. Cite evidence inline using [S1], [S2], etc.
6. If evidence is insufficient, say so.
7. Do not cite a source that does not support the claim.
8. Keep the answer concise but explain calculations when relevant.

QUESTION:
{question}

DETERMINISTIC CALCULATIONS:
{json.dumps(calculations, indent=2, default=str)}

EVIDENCE:
{"".join(evidence_blocks)}
"""

    response = client.models.generate_content(
        model=MODEL,
        contents=prompt,
        config=types.GenerateContentConfig(
            temperature=0,
        ),
    )

    return response.text.strip()

"""

Purpose
-------
Build a reliable, period-aware financial_facts_final table from SEC's

       SEC_USER_AGENT=Your Name your.email@example.com

class SECClient:
    BASE = "https://data.sec.gov"
    ARCHIVES = "https://www.sec.gov/Archives/edgar/data"

    def __init__(self, user_agent: str):
        self.session = requests.Session()
        self.session.headers.update(
            {
                "User-Agent": user_agent,
                "Accept-Encoding": "gzip, deflate",
                "Host": "data.sec.gov",
            }
        )

    def get_json(self, url: str) -> Dict[str, Any]:
        # Be polite to SEC infrastructure.
        time.sleep(0.25)

        response = self.session.get(url, timeout=45)
        response.raise_for_status()
        return response.json()

    def submissions(self, cik: str) -> Dict[str, Any]:
        return self.get_json(
            f"{self.BASE}/submissions/CIK{cik}.json"
        )

    def companyfacts(self, cik: str) -> Dict[str, Any]:
        return self.get_json(
            f"{self.BASE}/api/xbrl/companyfacts/CIK{cik}.json"
        )

def find_target_10q(
    submissions: Dict[str, Any],
    target_end: str,
    fiscal_quarter: str,
) -> Dict[str, str]:
    """
    Find the 10-Q whose reportDate equals the requested fiscal period end.

    Returns:
        accession, accession_no_dash, filing_date, report_date, primary_document
    """
    recent = submissions.get("filings", {}).get("recent", {})

    fields = [
        "form",
        "filingDate",
        "reportDate",
        "accessionNumber",
        "primaryDocument",
        "primaryDocDescription",
        "isXBRL",
    ]

    rows = []
    for field in fields:
        rows.append(recent.get(field, []))

    if not rows or not rows[0]:
        raise RuntimeError("SEC submissions response contains no recent filings.")

    records = [
        dict(zip(fields, values))
        for values in zip(*rows)
    ]

    candidates = [
        r
        for r in records
        if r["form"] == "10-Q"
        and r["reportDate"] == target_end
    ]

    if not candidates:
        # Some issuers may have a report-date representation that differs
        # slightly. Give a useful diagnostic rather than silently selecting
        # the wrong filing.
        nearby = [
            (
                r["form"],
                r["filingDate"],
                r["reportDate"],
                r["accessionNumber"],
                r["primaryDocument"],
            )
            for r in records
            if r["form"] == "10-Q"
        ][:20]

        raise RuntimeError(
            f"Could not find a 10-Q for {fiscal_quarter} ending {target_end}. "
            f"Recent 10-Qs seen: {nearby}"
        )

    # Prefer the XBRL filing if several candidates exist.
    candidates.sort(
        key=lambda r: (
            str(r.get("isXBRL", "0")) == "1",
            r["filingDate"],
        ),
        reverse=True,
    )

    selected = candidates[0]
    accession = selected["accessionNumber"]
    return {
        "accession": accession,
        "accession_no_dash": accession.replace("-", ""),
        "filing_date": selected["filingDate"],
        "report_date": selected["reportDate"],
        "primary_document": selected["primaryDocument"],
        "is_xbrl": str(selected.get("isXBRL", "")),
    }

def iso_days(start: str, end: str) -> int:
    return (date.fromisoformat(end) - date.fromisoformat(start)).days


def duration_months_from_days(days: int) -> Optional[int]:
    """
    SEC quarterly filings commonly use approximately:
      3 months = ~80-110 days
      6 months = ~170-200 days
      9 months = ~260-290 days
    """
    if 80 <= days <= 110:
        return 3
    if 170 <= days <= 200:
        return 6
    if 260 <= days <= 290:
        return 9
    if 350 <= days <= 380:
        return 12
    return None


def unit_to_display(unit: str, value: float) -> Tuple[str, float]:
    """
    Normalize USD values to USD millions because SEC financial statements
    typically present these statements in millions.

    Other units are retained as-is.
    """
    if unit == "USD":
        return "USD millions", value / 1_000_000.0

    if unit == "USD/shares":
        return "USD/share", value

    if unit == "shares":
        return "shares", value

    if unit == "pure":
        return "pure", value

    return unit, value


def iter_concept_facts(
    companyfacts: Dict[str, Any],
    concept: str,
) -> Iterable[Tuple[str, Dict[str, Any]]]:
    """
    Yield (unit, fact) for a us-gaap concept.
    """
    facts = companyfacts.get("facts", {})
    us_gaap = facts.get("us-gaap", {})
    concept_data = us_gaap.get(concept)

    if not concept_data:
        return

    units = concept_data.get("units", {})
    for unit_name, fact_rows in units.items():
        for fact in fact_rows:
            yield unit_name, fact


def choose_facts(
    companyfacts: Dict[str, Any],
    metric: str,
    target_end: str,
    accession: str,
    fiscal_year: int,
    fiscal_quarter: str,
) -> List[Dict[str, Any]]:
    """
    Return the current-quarter and prior-year-quarter XBRL facts for a metric.

    Important:
    - Q2 filings contain both 3-month and 6-month facts.
    - Q3 filings contain both 3-month and 9-month facts.
    - We keep ONLY ~3-month facts here.
    - We select facts by the filing accession, so the current filing's
      comparative prior-year values can also be captured.
    """
    target_date = date.fromisoformat(target_end)
    concepts = METRIC_ALIASES[metric]

    for concept in concepts:
        candidates: List[Tuple[str, Dict[str, Any], int]] = []

        for unit_name, fact in iter_concept_facts(companyfacts, concept):
            if fact.get("form") != "10-Q":
                continue

            if fact.get("accn") != accession:
                continue

            fact_fp = str(fact.get("fp", "")).upper()
            if fact_fp != fiscal_quarter:
                continue

            end = fact.get("end")
            start = fact.get("start")

            if not end:
                continue

            # Balance sheet facts are instantaneous.
            if not start:
                # Only use the current period-end for balance sheet metrics.
                if end != target_end:
                    continue

                candidates.append((unit_name, fact, 0))
                continue

            try:
                days = iso_days(start, end)
            except Exception:
                continue

            # Keep quarterly duration only.
            if duration_months_from_days(days) != 3:
                continue

            end_date = date.fromisoformat(end)
            distance_from_current = abs((end_date - target_date).days)

            # Keep the current quarter and the comparative prior-year quarter.
            # A fiscal quarter can shift by a few days (notably NVIDIA).
            if distance_from_current > 370:
                continue

            candidates.append((unit_name, fact, days))

        if not candidates:
            continue

        # Prefer USD for monetary metrics.
        candidates.sort(
            key=lambda item: (
                item[0] == "USD",
                item[1].get("filed", ""),
            ),
            reverse=True,
        )

        # Deduplicate by end date. If multiple facts exist for the same
        # end date, retain the first preferred unit/concept.
        selected: List[Dict[str, Any]] = []
        seen_ends = set()

        for unit_name, fact, days in candidates:
            end = fact["end"]
            if end in seen_ends:
                continue

            seen_ends.add(end)
            selected.append(
                {
                    "concept": concept,
                    "unit_name": unit_name,
                    "fact": fact,
                    "duration_months": (
                        3 if fact.get("start") else None
                    ),
                }
            )

        # We expect current + comparative for P&L metrics. Sort chronologically.
        selected.sort(key=lambda item: item["fact"]["end"])

        return selected

    return []

def period_label(
    fiscal_year: int,
    fiscal_quarter: str,
) -> str:
    return f"{fiscal_year}-{fiscal_quarter}"


def prior_year_period(
    fiscal_year: int,
    fiscal_quarter: str,
) -> str:
    return f"{fiscal_year - 1}-{fiscal_quarter}"


def make_rows(
    company: str,
    fiscal_year: int,
    fiscal_quarter: str,
    target_end: str,
    metric: str,
    selected_facts: List[Dict[str, Any]],
    source_file: str,
) -> List[Dict[str, Any]]:
    rows: List[Dict[str, Any]] = []

    target_date = date.fromisoformat(target_end)

    for selected in selected_facts:
        fact = selected["fact"]
        unit_name = selected["unit_name"]

        raw_value = float(fact["val"])
        display_unit, display_value = unit_to_display(unit_name, raw_value)

        fact_end = date.fromisoformat(fact["end"])

        if fact_end == target_date:
            period = period_label(fiscal_year, fiscal_quarter)
        else:
            # The comparison fact belongs to the same fiscal quarter in the
            # prior year. This also handles NVIDIA's shifted fiscal dates.
            period = period_label(fiscal_year - 1, fiscal_quarter)

        rows.append(
            {
                "company": company,
                "fiscal_year": fiscal_year,
                "fiscal_quarter": fiscal_quarter,
                "period_end": fact["end"],
                "metric": metric,
                "source_label": metric.replace("_", " ").title(),
                "period": period,
                "duration_months": selected["duration_months"],
                "value": display_value,
                "unit": display_unit,
                "pdf_page": None,
                "section": None,
                "table_name": None,
                "source_file": source_file,
                "validation_source": "SEC_XBRL",
                "created_at": datetime.utcnow().isoformat(timespec="seconds") + "Z",
            }
        )

    return rows

def table_id() -> str:
    return f"{PROJECT_ID}.{DATASET}.{TABLE}"


def ensure_table(client: bigquery.Client) -> None:
    sql = f"""
    CREATE TABLE IF NOT EXISTS `{table_id()}`
    (
      company STRING,
      fiscal_year INT64,
      fiscal_quarter STRING,
      period_end DATE,

      metric STRING,
      source_label STRING,

      period STRING,
      duration_months INT64,

      value FLOAT64,
      unit STRING,

      pdf_page INT64,
      section STRING,
      table_name STRING,

      source_file STRING,

      validation_source STRING,

      created_at TIMESTAMP
    )
    """
    client.query(sql).result()


def insert_rows(client: bigquery.Client, rows: List[Dict[str, Any]]) -> None:
    if not rows:
        return

    errors = client.insert_rows_json(
        table_id(),
        rows,
        skip_invalid_rows=False,
        ignore_unknown_values=False,
    )

    if errors:
        raise RuntimeError(f"BigQuery insert errors: {errors}")

def extract() -> None:
    sec = SECClient(SEC_USER_AGENT)
    bq = bigquery.Client(project=PROJECT_ID)

    ensure_table(bq)

    # We rebuild this final table on each explicit extraction run.
    bq.query(f"TRUNCATE TABLE `{table_id()}`").result()

    all_rows: List[Dict[str, Any]] = []
    filing_count = 0

    for company, config in COMPANIES.items():
        cik = config["cik"]

        print(f"\nLoading SEC submissions: {company}")
        submissions = sec.submissions(cik)

        print(f"Loading SEC Company Facts: {company}")
        companyfacts = sec.companyfacts(cik)

        for fiscal_year, fiscal_quarter, target_end in TARGET_FILINGS[company]:
            filing = find_target_10q(
                submissions,
                target_end,
                fiscal_quarter,
            )

            accession = filing["accession"]
            source_file = (
                f"FY{fiscal_year}_{fiscal_quarter}_10Q.pdf"
            )

            filing_count += 1

            print(
                f"  {company.upper()} {fiscal_quarter} "
                f"FY{fiscal_year} | {target_end} | "
                f"{accession}"
            )

            for metric in METRIC_ALIASES:
                selected_facts = choose_facts(
                    companyfacts=companyfacts,
                    metric=metric,
                    target_end=target_end,
                    accession=accession,
                    fiscal_year=fiscal_year,
                    fiscal_quarter=fiscal_quarter,
                )

                if not selected_facts:
                    print(f"      WARNING: no XBRL fact for {metric}")
                    continue

                rows = make_rows(
                    company=company,
                    fiscal_year=fiscal_year,
                    fiscal_quarter=fiscal_quarter,
                    target_end=target_end,
                    metric=metric,
                    selected_facts=selected_facts,
                    source_file=source_file,
                )

                all_rows.extend(rows)

                for row in rows:
                    print(
                        f"      {metric}: "
                        f"{row['period']} = "
                        f"{row['value']} {row['unit']} "
                        f"({row['duration_months']} months)"
                    )

    insert_rows(bq, all_rows)

    print("\n" + "=" * 70)
    print("XBRL FINANCIAL FACT INGESTION COMPLETE")
    print("=" * 70)
    print(f"Filings processed : {filing_count}")
    print(f"Facts inserted    : {len(all_rows)}")
    print(f"BigQuery table    : {table_id()}")
    print("=" * 70)


if __name__ == "__main__":
    try:
        extract()
    except Exception as exc:
        print("\nXBRL EXTRACTION FAILED")
        print(f"{type(exc).__name__}: {exc}")
        sys.exit(1)

# SEC Financial Analyst Copilot

Complete reference implementation for a 12-filing SEC 10-Q RAG project.

## Data
Amazon FY2025 Q1-Q3; Alphabet FY2025 Q1-Q3; Meta FY2025 Q1-Q3; NVIDIA FY2026 Q1-Q3.

Put your existing PDFs in `data/<company>/` at the repository root. Do not rename NVIDIA FY2026 files.

## Pipeline
PDF -> page/table extraction -> metadata-rich chunks -> Gemini embeddings -> BigQuery VECTOR_SEARCH -> deterministic calculator -> Gemini grounded response -> citations -> React UI.

## Setup
1. Create/activate Python 3.11 virtualenv.
2. `pip install -r backend/requirements.txt`
3. Copy `backend/.env.example` to `backend/.env`.
4. Set `GOOGLE_CLOUD_PROJECT`.
5. `gcloud auth application-default login`.
6. Enable BigQuery and Vertex AI APIs.
7. Replace PROJECT_ID in `backend/sql/schema.sql` and run it.
8. From repo root run `python backend/app.py` once to ingest. It will parse PDFs, generate embeddings and load BigQuery.
9. Create the vector index after ingestion using the commented SQL.
10. Start API with `uvicorn backend.app:app --reload --port 8000`.
11. In frontend: `npm install && npm run dev`.

## Important limitation
The automatic financial-fact extractor is deliberately conservative. It only accepts table rows whose labels match known metrics and stores the numeric cells as candidates. Before a production deployment, validate column semantics (current vs prior period) against table headers and expand metric aliases. Do not claim numeric correctness from an unvalidated candidate fact.

## Evaluation
Test direct lookups, YoY/QoQ calculations, cross-company comparisons, narrative questions and an out-of-corpus question. Every numeric answer must be supported by a source chunk and a deterministic calculation.
